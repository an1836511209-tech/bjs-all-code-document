## 0. 服务健康检查

### GET /health (智能体健康检查)

检查智能体服务是否正常运行，用于探活和运维监控。

#### 请求示例

```http
GET /health HTTP/1.1
Host: localhost:8580
```

#### 响应示例 - 成功

```json
{
  "status": "healthy",
  "service": "北交所录入助审系统",
  "version": "1.0.0",
  "timestamp": "2025-10-22T10:00:00Z"
}
```

#### 响应示例 - 服务不可用

服务未启动时，客户端将收到连接错误，无 JSON 响应体。

## 1. 文件质量检测

```http
Host: api.example.com
Authorization: Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...
Content-Type: application/json
```

```javascript
class BeijiaoAPIClient {
  constructor(baseURL, token) {
    this.baseURL = baseURL;
    this.token = token;
  }

  // 文件质量检测
  async checkFileQuality(files) {
    const response = await fetch(`${this.baseURL}/file/quality-check`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${this.token}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({ files })
    });
    return await response.json();
  }

  // 同步字段提取
  async extractFieldsSync(extractionType, files, ruleConfigId, timeout = 1000) {
    const response = await fetch(`${this.baseURL}/extraction/sync`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${this.token}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        extractionType,
        files,
        ruleConfigId,
        timeout,
        requestId: `req_${Date.now()}`
      })
    });
    return await response.json();
  }

  // 流式字段提取
  async extractFieldsStream(extractionType, files, ruleConfigId, onEvent) {
    const response = await fetch(`${this.baseURL}/extraction/stream`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${this.token}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        extractionType,
        files,
        ruleConfigId
      })
    });

    const reader = response.body.getReader();
    const decoder = new TextDecoder();

    while (true) {
      const { done, value } = await reader.read();
      if (done) break;

      const chunk = decoder.decode(value);
      const lines = chunk.split('\n');

      for (const line of lines) {
        if (line.startsWith('event:')) {
          const eventType = line.substring(6).trim();
          const dataLine = lines[lines.indexOf(line) + 1];
          if (dataLine && dataLine.startsWith('data:')) {
            const data = JSON.parse(dataLine.substring(5).trim());
            onEvent(eventType, data);
          }
        }
      }
    }
  }

  // 查询任务状态
  async getTaskStatus(taskId, includeDetails = true) {
    const response = await fetch(
      `${this.baseURL}/task/${taskId}?includeDetails=${includeDetails}`,
      {
        headers: {
          'Authorization': `Bearer ${this.token}`
        }
      }
    );
    return await response.json();
  }
}

// 使用示例
const client = new BeijiaoAPIClient('https://api.example.com/api/v1', 'your-token');

// 文件质量检测

  }
]);

  } else if (eventType === 'task_complete') {

  }
});
```

### Python

```python
import requests
from typing import List, Dict, Callable

class BeijiaoAPIClient:
    def __init__(self, base_url: str, token: str):
        self.base_url = base_url
        self.token = token
        self.headers = {
            'Authorization': f'Bearer {token}',
            'Content-Type': 'application/json'
        }
    
    def check_file_quality(self, files: List[Dict]) -> Dict:
        """文件质量检测"""
        response = requests.post(
            f'{self.base_url}/file/quality-check',
            headers=self.headers,
            json={'files': files}
        )
        return response.json()
    
    def extract_fields_sync(
        self,
        extraction_type: str,
        files: List[Dict],
        rule_config_id: str = None,
        timeout: int = 1000
    ) -> Dict:
        """同步字段提取"""
        import time
        payload = {
            'extractionType': extraction_type,
            'files': files,
            'ruleConfigId': rule_config_id,
            'timeout': timeout,
            'requestId': f'req_{int(time.time() * 1000)}'
        }
        response = requests.post(
            f'{self.base_url}/extraction/sync',
            headers=self.headers,
            json=payload
        )
        return response.json()
    
    def extract_fields_stream(
        self,
        extraction_type: str,
        files: List[Dict],
        rule_config_id: str,
        on_event: Callable
    ):
        """流式字段提取"""
        payload = {
            'extractionType': extraction_type,
            'files': files,
            'ruleConfigId': rule_config_id
        }
        response = requests.post(
            f'{self.base_url}/extraction/stream',
            headers=self.headers,
            json=payload,
            stream=True
        )
        
        for line in response.iter_lines():
            if line:
                line = line.decode('utf-8')
                if line.startswith('event:'):
                    event_type = line[6:].strip()
                elif line.startswith('data:'):
                    import json
                    data = json.loads(line[5:].strip())
                    on_event(event_type, data)
    
    def get_task_status(self, task_id: str, include_details: bool = True) -> Dict:
        """查询任务状态"""
        response = requests.get(
            f'{self.base_url}/task/{task_id}',
            headers={'Authorization': f'Bearer {self.token}'},
            params={'includeDetails': include_details}
        )
        return response.json()

# 使用示例
client = BeijiaoAPIClient('https://api.example.com/api/v1', 'your-token')

# 文件质量检测
result = client.check_file_quality([
    {
        'fileUrl': 'https://example.com/file.pdf',
        'fileName': 'application.pdf',
        'fileType': 'pdf'
    }
])
print(result)

# 流式提取
def handle_event(event_type, data):
    if event_type == 'field_complete':
        print(f"字段提取完成: {data['fieldName']} = {data['extractedValue']}")
    elif event_type == 'task_complete':
        print(f"任务完成: {data}")

client.extract_fields_stream('大宗实物', files, 'rule_dzswv10', handle_event)
```
